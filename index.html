function New-Popup {
param ($Text = "", $Title = "", $button = "OK", $Icon = "None")

}

function Get-Elapsed {
param($var, $Format = "None")
if ($Format -eq "Milliseconds") {
return $var.ElapsedMilliseconds
} elseif ($Format -eq "Ticks") {
return $var.ElapsedTicks
} elseif ($Format -eq "None") {
return $var.Elapsed
} else {
Write-Host "
Use:
-Format 'Milliseconds'
-Format 'Ticks'
-Format 'None'
"
}
}
function Get-RunningProcessPath {
param($Process)
$x = Get-Process -Name $Process
return $x.Path
}

function New-Shortcut {
param($TargetPath, $ShortcutPath)
$WshShell = New-Object -ComObject WScript.Shell
$Shortcut = $WshShell.CreateShortcut($ShortcutPath)
$Shortcut.TargetPath = $TargetPath
$Shortcut.Save()
}
function Set-ShortcutIcon {
param($ShortcutPath, $NewIconPath)
$WshShell = New-Object -ComObject WScript.Shell
$Shortcut = $WshShell.CreateShortcut($ShortcutPath)
$Shortcut.IconLocation = $NewIconPath
$Shortcut.Save()
}
function Invoke-WebContent {
param($Uri)
$response = Invoke-WebRequest -Uri $Uri -UseBasicParsing
$doc = New-Object -ComObject 'HTMLFile'
$doc.IHTMLDocument2_write($response.Content)
return $doc.body.innerText
}
